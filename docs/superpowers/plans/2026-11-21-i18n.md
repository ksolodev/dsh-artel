# Full i18n Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Move every user-visible string in dsh-artel into per-language catalogs, so the UI, feed, journal, queue, Factory, errors and plugin metadata can switch between English, Russian, Chinese and any future language without a code change.

**Architecture:** `locale/<tag>.json` is the only source of translations. `tools/build-locales.mjs` compiles those catalogs into a generated `const DICTIONARIES` block inside the classic client script. The Host never imports translations; it emits `{ key, params, text }` records, and the client resolves the key against the active language with the legacy `text` as an English fallback. The active language lives in `<stateDir>/settings.json` and is exposed through `GET/POST /api/dsh-artel/settings`.

**Tech Stack:** Node.js ESM (no build step, no dependencies), plain CommonJS `require` inside the client's `window.__ModuleLoader__.load` factory, React via the Host runtime, `node:test` + `node:assert/strict`, `Intl.PluralRules` / `Intl.DateTimeFormat`.

**Spec:** `docs/superpowers/specs/2026-11-21-i18n-design.md`

## Global Constraints

- Node `^22.19 || >=24`; no new runtime or dev dependencies; no build step for the plugin itself.
- One JSON file per language in `locale/`; the file name is the BCP-47 tag (`en`, `ru`, `zh`, `pt-BR`).
- `locale/en.json` is the source of truth. Every non-plural key in every catalog must exist in English with the same `{placeholder}` set.
- Tags are validated with `^[A-Za-z]{2,8}(-[A-Za-z0-9]{1,8})*$`; an unknown tag or a tag without a catalog is rejected with HTTP 400.
- Plural keys use the suffixes `.zero .one .two .few .many .other`; each language defines exactly `Intl.PluralRules(tag).resolvedOptions().pluralCategories`.
- The generated region in `lib/client.js` is committed and must stay fresh: `node tools/build-locales.mjs --check` fails on drift.
- Agent prompts, agent replies, command output, diffs, commit/PR texts and file paths are NOT translated.
- Tests run with `npm test` (which also runs the locale check). Commit style: `feat:`, `fix:`, `test:`, `docs:`, `chore:`.

```js
// tools/migrate-locales.mjs — run once, then delete.
import fs from 'node:fs'
import path from 'node:path'

const ROOT = path.resolve(import.meta.dirname, '..')
const CLIENT = path.join(ROOT, 'lib', 'client.js')
const LOCALE = path.join(ROOT, 'locale')
const BEGIN = '/* @generated locales:begin */'
const END = '/* @generated locales:end */'
const NATIVE = { en: 'English', ru: 'Русский', zh: '中文' }
const MARKERS = [['en', '    const en = {'], ['ru', '    const ru = {'], ['zh', '    const zh = {']]

/** Return the object literal starting at `marker`, brace-matched, quotes-aware. */
function objectAt(source, marker) {
  const start = source.indexOf(marker)
  if (start < 0) throw new Error(`missing ${marker}`)
  const open = source.indexOf('{', start)
  let depth = 0
  let quote = null
  for (let index = open; index < source.length; index += 1) {
    const ch = source[index]
    if (quote !== null) {
      if (ch === '\\') { index += 1; continue }
      if (ch === quote) quote = null
      continue
    }
    if (ch === "'" || ch === '"' || ch === '`') { quote = ch; continue }
    if (ch === '{') depth += 1
    else if (ch === '}') {
      depth -= 1
      if (depth === 0) return source.slice(open, index + 1)
    }
  }
  throw new Error(`unbalanced object for ${marker}`)
}

const source = fs.readFileSync(CLIENT, 'utf8')
for (const [tag, marker] of MARKERS) {
  const strings = Function(`"use strict"; return (${objectAt(source, marker)})`)()
  const current = JSON.parse(fs.readFileSync(path.join(LOCALE, `${tag}.json`), 'utf8'))
  const catalog = {
    meta: { ...current.meta, nativeName: NATIVE[tag], dir: current.meta?.dir ?? 'ltr' },
    strings,
  }
  fs.writeFileSync(path.join(LOCALE, `${tag}.json`), JSON.stringify(catalog, null, 2) + '\n', 'utf8')
  console.log(`${tag}: ${Object.keys(strings).length} strings`)
}

const anchorStart = source.indexOf('    const en = {')
const dictStart = source.indexOf('    const DICTIONARIES = { en, ru, zh }')
if (anchorStart < 0 || dictStart < 0) throw new Error('legacy dictionary layout not found')
const dictEnd = source.indexOf('\n', dictStart)
const block = `    ${BEGIN}\n    const DICTIONARIES = {}\n    ${END}`
fs.writeFileSync(CLIENT, source.slice(0, anchorStart) + block + source.slice(dictEnd), 'utf8')
console.log('lib/client.js: dictionaries replaced by the generated region')
```

```js
#!/usr/bin/env node
/**
 * Compiles locale/*.json into the generated dictionary block of lib/client.js.
 * Usage: node tools/build-locales.mjs [--check]
 */
import fs from 'node:fs'
import path from 'node:path'
import { fileURLToPath } from 'node:url'

const ROOT = path.resolve(path.dirname(fileURLToPath(import.meta.url)), '..')
const LOCALE_DIR = path.join(ROOT, 'locale')
const CLIENT_PATH = path.join(ROOT, 'lib', 'client.js')
export const BEGIN = '/* @generated locales:begin */'
export const END = '/* @generated locales:end */'
const PLURAL_SUFFIX = /^(.*)\.(zero|one|two|few|many|other)$/
const TAG = /^[A-Za-z]{2,8}(-[A-Za-z0-9]{1,8})*$/
const BASE = 'en'

export function readCatalogs(localeDir = LOCALE_DIR) {
  const catalogs = {}
  for (const file of fs.readdirSync(localeDir).filter((name) => name.endsWith('.json')).sort()) {
    const tag = file.slice(0, -'.json'.length)
    if (!TAG.test(tag)) throw new Error(`${file}: "${tag}" is not a BCP-47 language tag`)
    const parsed = JSON.parse(fs.readFileSync(path.join(localeDir, file), 'utf8'))
    if (parsed === null || typeof parsed !== 'object') throw new Error(`${file}: expected an object`)
    if (parsed.meta === null || typeof parsed.meta !== 'object') throw new Error(`${file}: meta is missing`)
    if (typeof parsed.meta.nativeName !== 'string' || parsed.meta.nativeName.trim() === '') throw new Error(`${file}: meta.nativeName is required`)
    if (parsed.meta.dir !== undefined && parsed.meta.dir !== 'ltr' && parsed.meta.dir !== 'rtl') throw new Error(`${file}: meta.dir must be "ltr" or "rtl"`)
    if (parsed.strings === null || typeof parsed.strings !== 'object') throw new Error(`${file}: strings is missing`)
    for (const [key, value] of Object.entries(parsed.strings)) {
      if (typeof value !== 'string' || value.trim() === '') throw new Error(`${file}: "${key}" must be a non-empty string`)
    }
    catalogs[tag] = parsed
  }
  if (catalogs[BASE] === undefined) throw new Error('locale/en.json is required: English is the fallback')
  return catalogs
}

function pluralBase(key) {
  const match = PLURAL_SUFFIX.exec(key)
  return match === null ? null : match[1]
}

function placeholders(value) {
  return [...String(value).matchAll(/\{(\w+)\}/g)].map((match) => match[1]).sort().join(',')
}

function validate(catalogs) {
  const english = catalogs[BASE].strings
  const englishKeys = Object.keys(english).filter((key) => pluralBase(key) === null).sort()
  const bases = new Set()
  for (const catalog of Object.values(catalogs)) {
    for (const key of Object.keys(catalog.strings)) {
      const base = pluralBase(key)
      if (base !== null) bases.add(base)
    }
  }
  for (const [tag, catalog] of Object.entries(catalogs)) {
    const strings = catalog.strings
    const ownKeys = Object.keys(strings).filter((key) => pluralBase(key) === null)
    const missing = englishKeys.filter((key) => !ownKeys.includes(key))
    const extra = ownKeys.filter((key) => english[key] === undefined)
    if (missing.length > 0) throw new Error(`${tag}: missing keys: ${missing.join(', ')}`)
    if (extra.length > 0) throw new Error(`${tag}: keys not in en: ${extra.join(', ')}`)
    for (const key of englishKeys) {
      if (placeholders(strings[key]) !== placeholders(english[key])) throw new Error(`${tag}: placeholders differ for ${key}`)
    }
    const required = new Set(new Intl.PluralRules(tag).resolvedOptions().pluralCategories)
    for (const base of bases) {
      const forms = new Set(Object.keys(strings)
        .filter((key) => pluralBase(key) === base)
        .map((key) => key.slice(base.length + 1)))
      for (const category of required) {
        if (!forms.has(category)) throw new Error(`${tag}: ${base}.${category} is missing`)
      }
      for (const form of forms) {
        if (!required.has(form)) throw new Error(`${tag}: ${base}.${form} is not a plural category of ${tag}`)
      }
      const reference = english[`${base}.other`] ?? english[`${base}.one`] ?? strings[`${base}.other`] ?? strings[`${base}.one`]
      if (reference !== undefined) {
        for (const form of forms) {
          const key = `${base}.${form}`
          if (placeholders(strings[key]) !== placeholders(reference)) throw new Error(`${tag}: placeholders differ for ${key}`)
        }
      }
    }
  }
}

export function render(catalogs) {
  const tags = Object.keys(catalogs).sort((a, b) => a === BASE ? -1 : b === BASE ? 1 : a.localeCompare(b))
  const lines = [BEGIN, '    const DICTIONARIES = {']
  for (const tag of tags) {
    lines.push(`      ${JSON.stringify(tag)}: {`)
    for (const key of Object.keys(catalogs[tag].strings).sort()) {
      lines.push(`        ${JSON.stringify(key)}: ${JSON.stringify(catalogs[tag].strings[key])},`)
    }
    lines.push('      },')
  }
  lines.push('    }', END)
  return lines.join('\n')
}

export function replaceRegion(source, block) {
  const begin = source.indexOf(BEGIN)
  const end = source.indexOf(END)
  if (begin < 0 || end < 0 || end < begin) throw new Error(`lib/client.js has no ${BEGIN} … ${END} region`)
  return source.slice(0, begin) + block + source.slice(end + END.length)
}

function main() {
  const check = process.argv.includes('--check')
  const catalogs = readCatalogs()
  validate(catalogs)
  const source = fs.readFileSync(CLIENT_PATH, 'utf8')
  const next = replaceRegion(source, render(catalogs))
  if (check) {
    if (next !== source) {
      console.error('lib/client.js locale block is stale; run: node tools/build-locales.mjs')
      process.exit(1)
    }
    console.log('locale block is up to date')
    return
  }
  if (next !== source) {
    fs.writeFileSync(CLIENT_PATH, next, 'utf8')
    console.log('updated lib/client.js locale block')
  } else {
    console.log('locale block is already up to date')
  }
}

if (process.argv[1] !== undefined && path.resolve(process.argv[1]) === fileURLToPath(import.meta.url)) main()
```

```js
import fs from 'node:fs/promises'
import path from 'node:path'

export const SETTINGS_VERSION = 1
const LANGUAGE_TAG = /^[A-Za-z]{2,8}(-[A-Za-z0-9]{1,8})*$/

export function isValidLanguageTag(value) {
  return typeof value === 'string' && LANGUAGE_TAG.test(value.trim())
}

export function settingsPath(stateDir) {
  return path.join(stateDir, 'settings.json')
}

export function defaultSettings() {
  return { version: SETTINGS_VERSION, language: null }
}

/** Settings of one profile; a missing or unreadable file means "all defaults". */
export async function readSettings(stateDir) {
  let raw
  try {
    raw = await fs.readFile(settingsPath(stateDir), 'utf8')
  } catch {
    return defaultSettings()
  }
  let parsed
  try {
    parsed = JSON.parse(raw)
  } catch {
    return defaultSettings()
  }
  if (parsed === null || typeof parsed !== 'object') return defaultSettings()
  return {
    version: SETTINGS_VERSION,
    language: isValidLanguageTag(parsed.language) ? parsed.language : null,
  }
}

async function writeAtomic(file, text) {
  await fs.mkdir(path.dirname(file), { recursive: true })
  const temp = `${file}.${process.pid}.${Date.now()}.tmp`
  await fs.writeFile(temp, text, 'utf8')
  try {
    await fs.rename(temp, file)
  } catch (error) {
    // Windows refuses to replace an open file; removing first is the portable fix.
    if (process.platform !== 'win32') throw error
    await fs.rm(file, { force: true })
    await fs.rename(temp, file)
  }
}

/** Merge `patch` into the current settings and persist atomically. */
export async function writeSettings(stateDir, patch) {
  const current = await readSettings(stateDir)
  const next = { ...current, ...patch, version: SETTINGS_VERSION }
  if (next.language !== null && next.language !== undefined && !isValidLanguageTag(next.language)) {
    throw new Error(`invalid language tag: ${next.language}`)
  }
  next.language = next.language ?? null
  await writeAtomic(settingsPath(stateDir), JSON.stringify(next, null, 2) + '\n')
  return next
}

/** Every `<tag>.json` in `localeDir`, English first, with its native name. */
export async function availableLocales(localeDir) {
  const entries = await fs.readdir(localeDir, { withFileTypes: true })
  const catalogues = []
  for (const entry of entries) {
    if (!entry.isFile() || !entry.name.endsWith('.json')) continue
    const id = entry.name.slice(0, -'.json'.length)
    if (!isValidLanguageTag(id)) continue
    let meta = {}
    try {
      meta = JSON.parse(await fs.readFile(path.join(localeDir, entry.name), 'utf8')).meta ?? {}
    } catch {
      continue
    }
    catalogues.push({
      id,
      nativeName: typeof meta.nativeName === 'string' && meta.nativeName.trim() !== '' ? meta.nativeName : id,
      dir: meta.dir === 'rtl' ? 'rtl' : 'ltr',
    })
  }
  return catalogues.sort((left, right) => {
    if (left.id === 'en') return -1
    if (right.id === 'en') return 1
    return left.id.localeCompare(right.id)
  })
}
```

      const next = { ...languageState, ...patch }
      const changed = next.chosen !== languageState.chosen
        || next.dsh !== languageState.dsh
        || next.effective !== languageState.effective
        || next.loaded !== languageState.loaded
        || next.error !== languageState.error

      if (!changed) return
      languageState = next
      languageView = { ...languageState }
      for (const listener of Array.from(languageListeners)) {
        try {
          listener()
        } catch {
          /* one broken subscriber must not stop the others */
        }
      }

      languageListeners.add(listener)
      return () => languageListeners.delete(listener)
    }

    /** Re-render on a language change; the same pattern as useArtelStore. */
    function useLanguage() {
      const [view, setView] = useState(() => languageView)
      useEffect(() => {
        setView(languageView)
        return subscribeLanguage(() => setView(languageView))
      }, [])
      return view

    /** Exact tag → base language → English. */
    function resolveLanguage(tag) {
      const raw = typeof tag === 'string' ? tag.trim() : ''
      if (raw !== '' && DICTIONARIES[raw] !== undefined) return raw
      const base = raw.split('-')[0]
      if (base !== '' && DICTIONARIES[base] !== undefined) return base
      return 'en'
    }

    /** The language of the surrounding DSH shell, best effort. */
    function dshLanguageTag() {
      try {
        const tag = document?.documentElement?.lang
        if (typeof tag === 'string' && tag.trim() !== '') return tag.trim()
      } catch {
        /* no document while the module loads */
      }
      try {
        if (typeof navigator !== 'undefined' && typeof navigator.language === 'string' && navigator.language !== '') return navigator.language
      } catch {
        /* no navigator in some hosts */
      }
      return 'en'
    }

    function pluralSuffix(language, count) {
      let rules = pluralRules.get(language)
      if (rules === undefined) {
        rules = new Intl.PluralRules(language)
        pluralRules.set(language, rules)
      }
      return rules.select(Number.isFinite(Number(count)) ? Number(count) : 0)
    }

    function templateFor(language, key, params) {
      const table = DICTIONARIES[language] ?? DICTIONARIES.en
      const direct = table[key]
      if (direct !== undefined) return direct
      const base = key.replace(/\.(zero|one|two|few|many|other)$/, '')
      const count = params?.n ?? params?.count ?? 0
      return table[`${base}.${pluralSuffix(language, count)}`] ?? table[`${base}.other`]
    }

    /** Translate with English fallback; a missing key is logged once in dev. */
    function translateNow(key, params) {
      let value = templateFor(languageState.effective, key, params)
      if (value === undefined && languageState.effective !== 'en') value = templateFor('en', key, params)
      if (value === undefined) {
        if (!missingLanguageKeys.has(key)) {
          missingLanguageKeys.add(key)
          if (typeof console !== 'undefined' && typeof console.warn === 'function') console.warn(`dsh-artel: missing locale key ${key}`)
        }
        return key
      }
      return fill(value, params ?? {})
    }

      if (languageState.loaded) return Promise.resolve(languageView)
      if (languageInflight !== null) return languageInflight
      const dsh = dshLanguageTag()
      publishLanguage({ dsh, effective: resolveLanguage(languageState.chosen ?? dsh) })
      languageInflight = getApi().settings()
        .then((payload) => {
          const chosen = payload !== null && payload !== undefined && typeof payload.language === 'string' ? payload.language : null
          const available = Array.isArray(payload?.available)
            ? payload.available.filter((entry) => entry !== null && typeof entry?.id === 'string' && entry.id !== '')
            : []
          publishLanguage({
            chosen,
            available,
            loaded: true,
            effective: resolveLanguage(chosen ?? languageState.dsh),
            error: null,
          })
          return languageView
        })
        .catch((error) => {
          publishLanguage({ loaded: true, error: error instanceof Error ? error.message : String(error) })
          return languageView
        })
        .finally(() => { languageInflight = null })
      return languageInflight
    }

    function setLanguage(language) {
      const previous = languageState.chosen
      const chosen = language === null || language === undefined || language === '' ? null : String(language)
      publishLanguage({ chosen, effective: resolveLanguage(chosen ?? languageState.dsh), error: null })
      return getApi().setLanguage(chosen)
        .then((payload) => {
          const stored = payload !== null && payload !== undefined && typeof payload.language === 'string' ? payload.language : null
          publishLanguage({ chosen: stored, effective: resolveLanguage(stored ?? languageState.dsh), error: null })
          return languageView
        })
        .catch((error) => {
          publishLanguage({ chosen: previous, effective: resolveLanguage(previous ?? languageState.dsh), error: error instanceof Error ? error.message : String(error) })
          throw error
        })
    }

    const languageStore = {
      get: () => languageView,
      subscribe: subscribeLanguage,
      init: initLanguage,
      set: setLanguage,
      t: translateNow,
    }

    /** A record may carry `<field>Key`/`<field>Params` next to its legacy text. */
    function localized(t, record, field) {
      if (record === null || record === undefined) return ''
      const name = field ?? 'text'
      const key = record[name + 'Key']
      if (typeof key === 'string' && key !== '') return t(key, record[name + 'Params'] ?? {})
      const text = record[name]
      return text === undefined || text === null ? '' : String(text)
    }

    /** Feed events use the compact `{ key, params, text }` shape. */
    function localizedEvent(t, event) {
      if (event === null || event === undefined) return ''
      if (typeof event.key === 'string' && event.key !== '') return t(event.key, event.params ?? {})
      return event.text === undefined || event.text === null ? '' : String(event.text)
    }
